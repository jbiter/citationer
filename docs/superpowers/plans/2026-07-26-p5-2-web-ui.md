# P5-2 Web UI（`citationer serve`）实现计划

> **给执行代理的说明：** 实现本计划时，请使用 `superpowers:subagent-driven-development`（推荐）或 `superpowers:executing-plans` 按任务逐步执行。每个步骤使用复选框 `- [ ]` 进行跟踪。

**目标：** 新增本地 Web 服务器命令 `citationer serve`，通过浏览器仪表板暴露 CLI 的分析能力，首期优先覆盖 PRD 中指定的 stats、network、compare 三大模块。

**架构：** 后端使用 FastAPI（`src/citationer/web/`）提供 JSON 接口和静态图表资源；前端使用原生 JS + Plotly（`src/citationer/web/static/`）渲染仪表板。服务端复用 CLI 现有的 `get_records()` 和分析引擎，确保 Web UI 与 CLI 逻辑保持一致。

**技术栈：** FastAPI、Uvicorn、Jinja2（首页模板）、Plotly（前端图表）。新增的服务器依赖全部放在 `[web]` optional-extra 组中，保持 CLI 本身轻量。

## 全局约束

- Python >= 3.11
- 必须复用 `src/citationer/analysis/` 中已有的分析引擎，不得重写分析逻辑
- 数据库路径遵循 CLI 约定：`Path.cwd() / ".citationer" / "cache.db"`
- Web 依赖为可选；未安装 `[web]` 时，`citationer serve` 必须以友好提示失败
- 现有分析引擎中的重依赖（`networkx`、`plotly`、`matplotlib` 等）均为惰性导入，Web 层保持此模式
- 项目测试覆盖率需维持 >= 80%
- 代码风格遵循现有 CLI：类型注解、适时使用 Pydantic、Typer 管理新命令

from fastapi import Depends, HTTPException

from citationer.utils.config import get_db_path
from citationer.utils.database import CitationDatabase
from citationer.utils.db_loader import load_records_from_db


def _db_path() -> Path:
    return get_db_path()


def get_db() -> CitationDatabase:
    """返回当前工作目录下已初始化的 CitationDatabase。"""
    db = CitationDatabase(_db_path())
    db.initialize()
    return db


def get_records() -> list:
    """从本地数据库加载记录；未导入时返回 400。"""
    records = load_records_from_db(_db_path())
    if not records:
        raise HTTPException(
            status_code=400,
            detail="尚未导入记录。请先运行 'citationer import'。",

    return records

"""JSON 安全序列化辅助函数。"""

from __future__ import annotations

import dataclasses
from pathlib import Path
from typing import Any


def to_json_safe(obj: Any) -> Any:
    """递归转换 dataclass、Path、元组键字典为 JSON 安全类型。"""
    if dataclasses.is_dataclass(obj):
        return to_json_safe(dataclasses.asdict(obj))
    if isinstance(obj, Path):
        return str(obj)
    if isinstance(obj, tuple):
        return list(obj)
    if isinstance(obj, dict):
        return {_json_key(k): to_json_safe(v) for k, v in obj.items()}
    if isinstance(obj, list):
        return [to_json_safe(v) for v in obj]
    return obj


def _json_key(key: Any) -> str:
    if isinstance(key, tuple):
        return "|".join(str(part) for part in key)
    return str(key)
```

"""FastAPI 应用工厂。"""

from __future__ import annotations

from pathlib import Path

from fastapi import FastAPI
from fastapi.middleware.cors import CORSMiddleware
from fastapi.staticfiles import StaticFiles

from citationer import __version__
from citationer.web.routers import charts, compare, data, network, stats


def create_app() -> FastAPI:
    app = FastAPI(
        title="Citationer",
        description="本地文献题录分析仪表板",
        version=__version__,
    )
    app.add_middleware(
        CORSMiddleware,
        allow_origins=["*"],
        allow_credentials=False,
        allow_methods=["*"],
        allow_headers=["*"],
    )

    app.include_router(stats.router, prefix="/api/stats", tags=["stats"])
    app.include_router(network.router, prefix="/api/network", tags=["network"])
    app.include_router(compare.router, prefix="/api/compare", tags=["compare"])
    app.include_router(data.router, prefix="/api/data", tags=["data"])
    app.include_router(charts.router, prefix="/api/charts", tags=["charts"])

    static_dir = Path(__file__).parent / "static"
    if static_dir.exists():
        app.mount("/", StaticFiles(directory=static_dir, html=True), name="static")

    return app
```

"""`citationer serve` — 启动本地 Web 仪表板。"""

from __future__ import annotations

import typer

app = typer.Typer(
    name="serve",
    help="启动本地 Web 仪表板。",
    no_args_is_help=False,
)


def serve(
    host: str = typer.Option("127.0.0.1", "--host", help="绑定主机"),
    port: int = typer.Option(8000, "--port", help="绑定端口"),
    reload: bool = typer.Option(False, "--reload", help="开发模式：自动重载"),
) -> None:
    """运行 Citationer Web 服务器。"""
    try:
        from citationer.web.app import create_app
        import uvicorn
    except ImportError as exc:
        typer.echo(
            "Web 服务器依赖未安装。请运行：pip install 'citationer[web]'",
            err=True,
        )
        raise typer.Exit(code=1) from exc

    uvicorn.run(
        "citationer.web.app:create_app",
        host=host,
        port=port,
        reload=reload,
        factory=True,
    )


if __name__ == "__main__":
    app()
```

"""多数据集对比 API 路由。"""

from __future__ import annotations

from typing import Annotated

from fastapi import APIRouter, Query

from citationer.analysis.compare import CompareEngine
from citationer.web.dependencies import RecordsDep
from citationer.web.serializers import to_json_safe

router = APIRouter()


def _engine(records: list, by: str) -> CompareEngine:
    return CompareEngine(records, by=by)


@router.get("/overview")
def overview(
    records: RecordsDep,
    by: Annotated[str, Query(pattern="^(database|file)$")] = "database",
    top_n: Annotated[int, Query(ge=1, le=200)] = 10,
    threshold: Annotated[float, Query(ge=0.0, le=1.0)] = 0.85,
) -> dict:
    overviews, overlaps = _engine(records, by).overview(top_n=top_n, threshold=threshold)
    return to_json_safe({
        "overviews": {k: v for k, v in overviews.items()},

"""图表生成 API 路由。"""

from __future__ import annotations

import tempfile
from pathlib import Path
from typing import Annotated

from fastapi import APIRouter, HTTPException, Path, Query
from fastapi.responses import FileResponse

from citationer.analysis.stats import StatsEngine
from citationer.web.dependencies import RecordsDep

router = APIRouter()


@router.get("/yearly.png")
def yearly_chart(
    records: RecordsDep,
    cumulative: bool = False,
) -> FileResponse:
    try:
        from citationer.viz.charts import generate_yearly_chart
    except ImportError as exc:
        raise HTTPException(status_code=501, detail="需要 viz extras") from exc

    with tempfile.NamedTemporaryFile(suffix=".png", delete=False) as tmp:
        generate_yearly_chart(records, Path(tmp.name), title="Yearly Publications", cumulative=cumulative)
        return FileResponse(tmp.name, media_type="image/png")


@router.get("/top/{kind}.png")
def top_chart(
    records: RecordsDep,
    kind: Annotated[str, Path(pattern="^(journals|authors|institutions)$")],
    top: Annotated[int, Query(ge=1, le=200)] = 20,
) -> FileResponse:
    try:
        from citationer.viz.charts import generate_top_n_chart
    except ImportError as exc:
        raise HTTPException(status_code=501, detail="需要 viz extras") from exc

    engine = StatsEngine(records)
    if kind == "journals":
        data = engine.journals(top_n=top)
        title, xlabel = "Top Journals", "Journal"
    elif kind == "authors":
        data = engine.authors(top_n=top)
        title, xlabel = "Top Authors", "Author"
    else:
        data = engine.institutions(top_n=top)
        title, xlabel = "Top Institutions", "Institution"

    with tempfile.NamedTemporaryFile(suffix=".png", delete=False) as tmp:
        generate_top_n_chart(data.items, Path(tmp.name), title=title, xlabel=xlabel, horizontal=True)
        return FileResponse(tmp.name, media_type="image/png")


@router.get("/network/{kind}.html")
def network_html(
    records: RecordsDep,
    kind: Annotated[str, Path(pattern="^(keywords|coauthors|cocitation|coupling)$")],
    top_n: Annotated[int, Query(ge=1, le=200)] = 50,
) -> FileResponse:
    try:
        from citationer.analysis.network import NetworkEngine
    except ImportError as exc:
        raise HTTPException(status_code=501, detail="需要 network extras") from exc

    engine = NetworkEngine(records)
    if kind == "keywords":
        result = engine.keyword_cooccurrence(top_n=top_n, threshold=3)
        edges, nodes = result.edges, result.keywords
    elif kind == "coauthors":
        result = engine.author_collaboration(min_papers=2, collab_type="authors")
        edges, nodes = result.edges, [n[0] for n in result.nodes]
    elif kind == "cocitation":
        result = engine.co_citation(top_n=top_n)
        edges, nodes = result.edges, []
    else:
        result = engine.bibliographic_coupling(top_n=top_n)
        edges, nodes = result.edges, []

    with tempfile.NamedTemporaryFile(suffix=".html", delete=False) as tmp:
        NetworkEngine.to_html(edges, nodes, getattr(result, "communities", {}), Path(tmp.name), title=f"{kind} network")
        return FileResponse(tmp.name, media_type="text/html")
```
